การตั้งค่าทั้งหมด

การติดตั้งและไลเซนส์

ทุกสิ่งที่คุณต้องการเพื่อเปิดใช้งาน resource ไลเซนส์ (key) เป็นส่วนหนึ่งของการติดตั้ง ดังนั้นจึงรวมไว้ที่นี่ทั้งหมด

1. แตกไฟล์ resource

วางโฟลเดอร์ MathStoreFairyWing-V7 ไว้ในโฟลเดอร์ resources/ ของคุณ

ชื่อโฟลเดอร์ชื่อโฟลเดอร์ต้องเป็น MathStoreFairyWing-V7 เท่านั้น Keybinds, เหตุการณ์ภายใน และ state bags ขึ้นอยู่กับชื่อนี้

2. ตั้งค่า Key (Convar)

แต่ละเซิร์ฟเวอร์ต้องการ key เฉพาะเพื่อให้ resource ทำงาน Key นี้ตั้งค่าเป็น convar ใน server.cfg ของคุณ และต้องอยู่ก่อน ensure

server.cfg
## ─── MathStore Fairy Wing V7 ───

# 1. กำหนด key (ก่อน ensure)
set mathstore_key_fairyv7 "AP-XXXX-XXXX-XXXX"

# 2. เริ่มต้น resource
ensure MathStoreFairyWing-V7
ระวังเรื่อง keyอย่าแชร์ key ของคุณ แต่ละ key ผูกกับ IP ของเซิร์ฟเวอร์ หากเปลี่ยนโฮสต์ ให้ติดต่อเพื่อย้าย key

3. Dependencies

Resourceสถานะฟังก์ชัน
ox_libไม่บังคับคำสั่งพร้อมคำแนะนำ, การแจ้งเตือนแบบกำหนดสไตล์ เปิดใช้งานใน Config.UseOxLib
qb-core / qbx_coreตรวจจับอัตโนมัติBridge QB/QBX สำหรับสิทธิ์และข้อมูลผู้เล่น
es_extendedตรวจจับอัตโนมัติBridge ESX สำหรับสิทธิ์และข้อมูลผู้เล่น
vrpตรวจจับอัตโนมัติBridge vRP สำหรับกลุ่ม, สิทธิ์ และตัวตน
โหมด Standaloneหากไม่พบ framework ใดๆ resource จะทำงานแบบอิสระ ทุกอย่างทำงานได้ เพียงแต่ไม่มีการเชื่อมต่อสิทธิ์ของ framework
การตั้งค่า

Config.lua

ไฟล์ config/config.lua รวมตัวเลือกทั้งหมดของ resource ทุกสิ่งที่คุณสามารถเปลี่ยนได้อยู่ที่นี่

Framework และทั่วไป

config.lua
-- ตรวจจับ framework อัตโนมัติ
Config.Framework = 'auto'     -- 'auto' | 'qbx' | 'qb' | 'esx' | 'vrp' | 'standalone'

-- ใช้ ox_lib สำหรับคำสั่งและการแจ้งเตือน
Config.UseOxLib = true

-- ภาษาของระบบ (การแจ้งเตือน, keybinds, HUD)
Config.Locale = 'pt-BR'      -- 'pt-BR' | 'en-US' | 'es' | 'fr' | 'pt-PT' | 'th'
เคล็ดลับเมื่อใช้ Config.Framework = 'auto' resource จะตรวจจับเองว่า framework ใดกำลังทำงาน บังคับด้วยตนเองเฉพาะเมื่อมีข้อขัดแย้ง

การแจ้งเตือน

config.lua
Config.Notifications = {
    type     = 'auto',    -- 'auto' | 'ox_lib' | 'qb' | 'esx' | 'vrp' | 'native'
    duration = 5000,     -- ระยะเวลาเป็นมิลลิวินาที
}
ค่าพฤติกรรม
'auto'ตรวจจับ: ox_lib > framework > native (ตามลำดับ)
'ox_lib'บังคับใช้ ox_lib:notify
'qb'บังคับใช้ QBCore notification
'esx'บังคับใช้ ESX.ShowNotification
'vrp'บังคับใช้ vRP notification
'native'การแจ้งเตือนแบบ native ของ GTA V

Keybinds

config.lua
Config.Keybinds = {
    toggle = '',      -- เปิด/ปิดปีก (ว่าง = ไม่มีปุ่มเริ่มต้น)
    fly    = false,    -- false = keybind ถูกปิดใช้งาน
    hud    = '',      -- HUD แบบวงกลม (ว่าง = ผู้เล่นตั้งค่าเอง)

วิธีการทำงานเมื่อค่าเป็น '' (ว่าง) จะไม่มีปุ่มใดถูกกำหนดโดยค่าเริ่มต้น ผู้เล่นไปที่ ตั้งค่า > แป้นพิมพ์ > FiveM และเลือกปุ่มที่ต้องการ ข้อความที่แสดงจะเปลี่ยนตาม Config.Locale

หลาย Framework

ระบบ Bridge

Resource ตรวจจับ framework ของเซิร์ฟเวอร์โดยอัตโนมัติและปรับฟังก์ชันภายในทั้งหมด ในกรณีส่วนใหญ่คุณไม่จำเป็นต้องตั้งค่าอะไร

วิธีการตรวจจับ

ใน bridge/shared.lua ระบบจะตรวจสอบว่า resources ใดกำลังทำงานและเลือก bridge ที่ถูกต้อง:

FrameworkResource ที่ตรวจพบผลลัพธ์
QBX (Qbox)qbx_core'qbx'
QB-Coreqb-core'qb'
ESXes_extended'esx'
vRPvrp'vrp'
ไม่มี-'standalone'

ฟังก์ชัน Bridge (ฝั่ง Server)

ฟังก์ชันที่มีอยู่ใน server สำหรับโต้ตอบกับข้อมูลผู้เล่น:

server
-- ดึงออบเจกต์ผู้เล่นจาก framework
local player = Bridge.GetPlayer(source)

-- ดึงตัวระบุเฉพาะ (citizenid, identifier, user_id)
local cid = Bridge.GetCitizenId(source)

-- ชื่อตัวละครใน framework
local name = Bridge.GetPlayerName(source)

-- อาชีพปัจจุบัน (ชื่อ + ระดับ)
local job = Bridge.GetJob(source)

-- ตรวจสอบสิทธิ์ (ระบบหลายชั้น)
local allowed = Bridge.HasPermission(source, 'use')

-- ส่งการแจ้งเตือน (ปรับตาม framework ที่ตรวจพบ)
Bridge.Notify(source, 'Mensagem', 'success')

-- เรียกใช้ callback ที่กำหนดในconfig
Bridge.ExecuteCallback('CanEquipWings', source, cor)

ฟังก์ชัน Bridge (ฝั่ง Client)

ฟังก์ชันที่มีอยู่ใน client:

client
-- ข้อมูลผู้เล่นจาก framework
local data = Bridge.GetPlayerData()

-- ผู้เล่นโหลดเสร็จแล้วหรือยัง?
local loaded = Bridge.IsPlayerLoaded()

-- การแจ้งเตือนฝั่ง client
Bridge.ShowNotification('Texto', 'info', 5000)

-- ox_lib พร้อมใช้งานในเซิร์ฟเวอร์หรือไม่?
local has = Bridge.HasOxLib()
การควบคุมการเข้าถึง

ระบบสิทธิ์การเข้าถึง

ควบคุมว่าใครสามารถใช้ฟังก์ชันแต่ละอย่างของ resource ได้ ระบบมี 8 ชั้นของการตรวจสอบ โดยตรวจสอบตามลำดับ เมื่อชั้นใดให้คำตอบที่ชัดเจน (อนุญาตหรือปฏิเสธ) ชั้นถัดไปจะไม่ถูกตรวจสอบ

คีย์สิทธิ์

มีสองคีย์ที่ใช้ทั่วทั้งระบบ:

คีย์ใช้ที่ไหนคำอธิบาย
'use'คำสั่งปีกทั้งหมด, HUD, แอนิเมชันสิทธิ์ทั่วไปในการใช้ resource
'admin_cleanup'/asaslimpar7สิทธิ์แอดมินสำหรับลบปีกทั้งหมดในเซิร์ฟเวอร์

ขั้นตอนการตรวจสอบทั้งหมด (8 ชั้น)

เมื่อผู้เล่นใช้คำสั่งใดๆ สิทธิ์จะถูกตรวจสอบตามลำดับนี้:

1. Config.Callbacks.HasPermission(source, permission) → คืนค่า true/false? ใช้ค่านี้และหยุดที่นี่ → คืนค่า nil? ดำเนินต่อไปยังชั้นถัดไป 2. ACE Permission: IsPlayerAceAllowed(source, Permissions.Ace[perm]) → เช่น: 'MathStoreFairyWing-V7.use' (สร้างอัตโนมัติ) 3. Feature Config: Permissions.Features[perm].enabled → ถ้า enabled = false ทุกคนมีสิทธิ์เข้าถึง (ไม่มีข้อจำกัด) 4. CitizenID: Permissions.Features[perm].citizenids → รายการ ID เฉพาะที่อนุญาต 5. vRP Groups: Permissions.VRP[perm] → เช่น: {'Fada', 'Admin'} 6. Sets (QBX): Permissions.Features[perm].sets 7. ACE Groups: Permissions.Groups[grupo] → ตรวจสอบ group.admin, group.vip, ฯลฯ 8. Jobs: Permissions.Features[perm].jobs → เช่น: {'police', 'ambulance'}

อนุญาตทุกคน (ไม่มีข้อจำกัดใดๆ)

permissions.lua
Permissions.Features = {
    use = {
        enabled = false,    -- false = ทุกคนใช้ได้ ไม่มีการตรวจสอบ
    },
}

จำกัดตามกลุ่ม vRP

permissions.lua
Permissions.VRP = {
    use = { 'Fada', 'Admin' },            -- ชื่อกลุ่ม vRP ที่ตรงกัน
    admin_cleanup = { 'Admin' },
}

จำกัดตาม ACE Permission

Resource สร้างชื่อ ACE โดยอัตโนมัติโดยใช้ GetCurrentResourceName() หากโฟลเดอร์คือ MathStoreFairyWing-V7 ACEs จะเป็น:

permissions.lua
Permissions.Ace = {
    use = GetCurrentResourceName() .. '.use',             -- 'MathStoreFairyWing-V7.use'
    admin_cleanup = GetCurrentResourceName() .. '.admin',  -- 'MathStoreFairyWing-V7.admin'
}

# ให้สิทธิ์ admin cleanup
add_ace group.admin MathStoreFairyWing-V7.admin allow

Callbacks

Callbacks: แทรกลอจิกของเซิร์ฟเวอร์คุณ

Callbacks ช่วยให้คุณเขียนลอจิกของคุณเองภายในขั้นตอนการทำงานของ resource โดยไม่ต้องแก้ไขไฟล์ภายใน ทั้งหมดอยู่ใน config/config.lua ใช้เพื่อตัดสินว่าใครสามารถใช้ได้ จะเกิดอะไรเมื่อสวมใส่ จะเกิดอะไรเมื่อถอดออก และกฎที่กำหนดเองของเซิร์ฟเวอร์คุณ

ตาราง Callbacks

Callbackพารามิเตอร์ค่าคืนเวลาที่เรียก
CanOpenHUDsourcetrue/falseก่อนเปิด HUD แบบวงกลม
CanEquipWingssource, cortrue/falseก่อนสวมปีก
OnWingsEquippedsource, corไม่มีหลังจากสวมปีก
OnWingsRemovedsourceไม่มีหลังจากถอดปีก
HasPermissionsource, permissiontrue/false/nilก่อนการตรวจสอบสิทธิ์ใดๆ
เกี่ยวกับ CanEquipWingsพารามิเตอร์ cor อาจเป็น nil หรือ number เมื่อเป็น nil server กำลังตรวจสอบว่าปุ่ม "รับปีก" ของ HUD ควรเปิดใช้งานหรือไม่ เมื่อเป็นตัวเลข จะเป็นความพยายามสวมจริง จัดการทั้งสองกรณีใน callback ของคุณ

สถานการณ์พร้อมตัวอย่างสมบูรณ์

แต่ละสถานการณ์ด้านล่างสมบูรณ์และเป็นอิสระ คัดลอกสิ่งที่เหมาะกับเซิร์ฟเวอร์ของคุณ

ต้องมีไอเท็มในกระเป๋าเพื่อสวม (และคืนเมื่อถอด) ESX

ผู้เล่นต้องมีไอเท็ม fairy_essence ในกระเป๋า เมื่อสวมใส่ ไอเท็มจะถูกใช้ เมื่อถอดปีก ไอเท็มจะกลับคืนสู่กระเป๋า

config.lua
Config.Callbacks = {
    CanEquipWings = function(source, cor)
        if cor then
            -- พยายามสวมจริง: ตรวจสอบไอเท็ม
            local xPlayer = ESX.GetPlayerFromId(source)
            local item = xPlayer.getInventoryItem('fairy_essence')
            if not item or item.count < 1 then
                Bridge.Notify(source, 'คุณต้องมี Fairy Essence!', 'error')
                return false
            end
            return true
        end
        -- cor = nil: แค่ตรวจสอบว่า HUD แสดงปุ่มหรือไม่
        return true
    end,

    OnWingsEquipped = function(source, cor)
        -- ลบไอเท็มจากกระเป๋าหลังสวมใส่
        local xPlayer = ESX.GetPlayerFromId(source)

เหตุการณ์และการเชื่อมต่อ

เหตุการณ์: การรับฟังและเชื่อมต่อ

Resource ส่งเหตุการณ์หลายรายการตลอดวงจรชีวิตของปีก คุณสามารถรับฟังเหตุการณ์เหล่านี้จาก resource อื่นเพื่อสร้างการเชื่อมต่อ ซิงโครไนซ์ระบบ หรือตอบสนองเมื่อมีเหตุการณ์เกิดขึ้น

การตั้งชื่อเหตุการณ์

เหตุการณ์ทั้งหมดมีคำนำหน้าด้วยชื่อ resource โดยอัตโนมัติเพื่อหลีกเลี่ยงข้อขัดแย้ง คำนำหน้าใช้รูปแบบ:

lua
-- รูปแบบ: ชื่อResource:ประเภท:การกระทำ
-- ตัวอย่าง:
'MathStoreFairyWing-V7:spawn'          -- คำสั่ง (server → client)
'MathStoreFairyWing-V7:sync:spawned'   -- ซิงโครไนซ์ (client → server)
'MathStoreFairyWing-V7:sync:spawn'     -- Broadcast (server → clients อื่น)
เกี่ยวกับชื่อชื่อที่แน่นอนถูกสร้างอัตโนมัติใน bridge/shared.lua โดยใช้ GetCurrentResourceName() ด้านล่างนี้ เราใช้ชื่อเริ่มต้น MathStoreFairyWing-V7

เหตุการณ์ Client → Server (ซิงโครไนซ์)

เมื่อผู้เล่นทำอะไรในเครื่อง client จะส่งเหตุการณ์เหล่านี้ไปยัง server เพื่อให้ผู้เล่นอื่นเห็น:

เหตุการณ์พารามิเตอร์เมื่อส่ง
MathStoreFairyWing-V7:sync:spawnedcor, initialStateปีกถูกสร้างในเครื่อง
MathStoreFairyWing-V7:sync:animChangedanimKeyแอนิเมชันปีกเปลี่ยน
MathStoreFairyWing-V7:sync:removed-ปีกถูกลบในเครื่อง
MathStoreFairyWing-V7:sync:requestBulk-ผู้เล่นขอสถานะปัจจุบัน (เมื่อเชื่อมต่อ)

เหตุการณ์ Server → Client (คำสั่ง)

เมื่อคำสั่งผ่านการตรวจสอบใน server (สิทธิ์ + callbacks) server จะส่งไปยัง client เพื่อดำเนินการ เหตุการณ์เหล่านี้ส่งไปยังผู้เล่นที่ใช้คำสั่ง:

เหตุการณ์พารามิเตอร์การกระทำ
MathStoreFairyWing-V7:spawncorสวมปีก
MathStoreFairyWing-V7:remove-ถอดปีก
MathStoreFairyWing-V7:toggle-เปิด/ปิดปีก
MathStoreFairyWing-V7:fly-สลับการบิน
MathStoreFairyWing-V7:changeColorcorเปลี่ยนสีปีก
MathStoreFairyWing-V7:hudcanEquipเปิด/ปิด HUD
MathStoreFairyWing-V7:deleteMassive-ลบทั้งหมด (admin, broadcast -1)
MathStoreFairyWing-V7:deleteNearby-ลบปีกที่ค้างใกล้เคียง
MathStoreFairyWing-V7:abrir-แอนิเมชันกางปีก
MathStoreFairyWing-V7:fechar-แอนิเมชันหุบปีก
MathStoreFairyWing-V7:bater-แอนิเมชันโบกปีก

เหตุการณ์ Broadcast (Server → ทุกคนใน Bucket)

เมื่อผู้เล่นสวมใส่/ถอดปีก server จะส่งไปยังผู้เล่นอื่นทั้งหมดใน routing bucket เดียวกัน:

เหตุการณ์พารามิเตอร์การกระทำ
MathStoreFairyWing-V7:sync:spawnserverId, cor, stateผู้เล่นอื่นสวมปีก
MathStoreFairyWing-V7:sync:animserverId, animKeyแอนิเมชันของผู้เล่นอื่นเปลี่ยน
MathStoreFairyWing-V7:sync:removeserverIdผู้เล่นอื่นถอดปีก
MathStoreFairyWing-V7:sync:bulkallWings (table)สถานะทั้งหมดของ bucket (ปีก)

เหตุการณ์ภายใน Client (รับฟังได้จาก resource อื่น)

Client ส่งเหตุการณ์ภายในโดยใช้ TriggerEvent (ไม่ใช่ net events เป็นเหตุการณ์ภายใน client) resource อื่นสามารถรับฟังเหตุการณ์เหล่านี้ด้วย AddEventHandler:

เหตุการณ์พารามิเตอร์เมื่อ
flight:startplayerIdผู้เล่นเริ่มบิน
flight:endplayerIdผู้เล่นหยุดบิน
flight:animationChangeplayerId, stateสถานะแอนิเมชันการบินเปลี่ยน
wing:rasanteActiveactive (bool)โหมดบินเรี่ยพื้นเปิด/ปิด
wing:subidaActiveactive (bool)โหมดบินขึ้นเปิด/ปิด

สถานการณ์การเชื่อมต่อพร้อมตัวอย่างสมบูรณ์

แต่ละสถานการณ์ด้านล่างแสดงวิธีรับฟังหรือตอบสนองต่อเหตุการณ์ของ resource จาก resource อื่นในเซิร์ฟเวอร์ของคุณ

ตอบสนองเมื่อผู้เล่นเริ่มหรือหยุดบิน Client - Resource อื่น

Resource HUD หรือระบบสถานะของคุณสามารถรับฟังจุดเริ่มต้น/สิ้นสุดการบิน เพื่อแสดงตัวบ่งชี้ ใช้เอฟเฟกต์ หรือเปลี่ยนกล้อง

client.lua (resource อื่น)
-- ผู้เล่นเริ่มบิน
AddEventHandler('flight:start', function(playerId)
    print('[MeuHUD] Player กำลังบิน: ' .. playerId)
    -- เปิดตัวบ่งชี้การบินใน HUD
    SendNUIMessage({ action = 'showFlightIndicator', active = true })
end)

-- ผู้เล่นหยุดบิน
AddEventHandler('flight:end', function(playerId)
    print('[MeuHUD] Player ลงพื้นแล้ว: ' .. playerId)
    SendNUIMessage({ action = 'showFlightIndicator', active = false })
end)

-- แอนิเมชันการบินเปลี่ยน (เช่น: ร่อน, กระพือปีก, ลดระดับ)
AddEventHandler('flight:animationChange', function(playerId, state)
    print('[MeuHUD] แอนิเมชันการบิน: ' .. state)
end)
ตรวจจับโหมดบินเฉพาะ (บินเรี่ยพื้น, บินขึ้น) Client - Resource อื่น

หากคุณต้องการใช้เอฟเฟกต์ภาพที่ต่างกันตามประเภทของท่าทางที่ผู้เล่นทำในอากาศ

client.lua (resource อื่น)
-- โหมดบินเรี่ยพื้น (บินต่ำความเร็วสูง)
AddEventHandler('wing:rasanteActive', function(active)
    if active then
        -- ใช้เอฟเฟกต์ความเร็ว, motion blur, ฯลฯ
        SetTimecycleModifier('spectator5')
    else
        ClearTimecycleModifier()
    end

รับฟังใน server เมื่อมีคนสวม/ถอดปีก Server - Resource อื่น

จาก resource อื่นใน server คุณสามารถรับฟังเหตุการณ์ sync เพื่อบันทึกล็อก ให้รางวัล แจ้ง Discord ฯลฯ

server.lua (resource อื่น)
-- รับฟังเมื่อมีคนสวมปีก
RegisterNetEvent('MathStoreFairyWing-V7:sync:spawned')
AddEventHandler('MathStoreFairyWing-V7:sync:spawned', function(cor, initialState)
    local src = source
    print('[Log] Player ' .. src .. ' สวมปีกสี ' .. cor)

    -- ตัวอย่าง: ส่งไปยัง Discord
    exports['discord_log']:sendLog('wings', 
        GetPlayerName(src) .. ' สวมปีก (สี ' .. cor .. ')'
    )
end)

-- รับฟังเมื่อมีคนถอดปีก
RegisterNetEvent('MathStoreFairyWing-V7:sync:removed')
AddEventHandler('MathStoreFairyWing-V7:sync:removed', function()
    local src = source
    print('[Log] Player ' .. src .. ' ถอดปีก')
end)
เกี่ยวกับการรับฟัง net eventsเมื่อคุณลงทะเบียน RegisterNetEvent จาก resource อื่น คุณกำลังรับฟังเหตุการณ์เดียวกับที่ Fairy Wing V7 รับฟังอยู่ภายใน FiveM อนุญาตให้มี handlers หลายตัว ลอจิกของคุณทำงานพร้อมกันโดยไม่รบกวน resource ต้นฉบับ

        end

ส่งเหตุการณ์ Fairy Wing V7 จาก resource อื่น

สถานการณ์ด้านบนแสดงวิธีรับฟังเหตุการณ์ แต่คุณยังสามารถส่งเหตุการณ์ของ Fairy Wing V7 โดยตรงจาก resource ของคุณ โดยไม่ต้องพึ่งคำสั่ง สิ่งนี้มีประโยชน์เมื่อขั้นตอนเริ่มต้นนอก Fairy Wing V7 (ร้านค้า, ไอเท็ม, NPC, เควส ฯลฯ)

สำคัญเมื่อคุณส่งเหตุการณ์โดยตรง คุณจะข้ามระบบสิทธิ์และ callbacks ของ Fairy Wing V7 ความรับผิดชอบในการตรวจสอบว่าผู้เล่นสามารถใช้ได้หรือไม่อยู่ที่ resource ของคุณ ใช้สิ่งนี้เฉพาะเมื่อคุณต้องการควบคุมขั้นตอนอย่างสมบูรณ์
ร้านค้าหรือ NPC ที่ให้ปีกเมื่อซื้อ Server - Resource อื่น

ผู้เล่นโต้ตอบกับ NPC หรือร้านค้า จ่ายเงิน และได้รับปีกโดยตรง จาก server ของ resource คุณ ส่งเหตุการณ์ :spawn ไปยัง client ของผู้เล่น

server.lua (resource อื่น - เช่น: ร้านค้า)
-- ผู้เล่นซื้อปีกในร้าน สี 5
RegisterNetEvent('minhaLoja:comprarAsas')
AddEventHandler('minhaLoja:comprarAsas', function(cor)
    local src = source

    -- ลอจิกการชำระเงินของคุณ
    local xPlayer = ESX.GetPlayerFromId(src)
    if xPlayer.getMoney() < 5000 then
        TriggerClientEvent('esx:showNotification', src, 'เงินไม่เพียงพอ!')
        return
    end
    xPlayer.removeMoney(5000)

    -- ให้ปีกแก่ผู้เล่น (ส่งโดยตรงไปยัง Fairy Wing V7)
    TriggerClientEvent('MathStoreFairyWing-V7:spawn', src, cor)
end)
ไอเท็มที่ใช้ได้เพื่อสวมปีกเมื่อใช้ ESX

ลบปีกของผู้เล่นจาก resource อื่น Server - Resource อื่น

หากผู้เล่นตาย ถูกจับ เข้าโซนจำกัด หรือเงื่อนไขอื่นของเซิร์ฟเวอร์คุณ คุณสามารถบังคับให้ลบปีกได้

ขั้นตอนทั้งหมด: จากคำสั่งถึงการซิงโครไนซ์

เพื่อทำความเข้าใจว่าทุกอย่างเชื่อมต่อกันอย่างไร นี่คือขั้นตอนทั้งหมดเมื่อผู้เล่นพิมพ์ /asas7 5:

ผู้เล่น: /asas7 5 ↓ Server: รับคำสั่งที่ลงทะเบียน ↓ Bridge.HasPermission(source, 'use') → ตรวจสอบ 8 ชั้น (ดูส่วนสิทธิ์การเข้าถึง) ↓ ไม่มีสิทธิ์? → Bridge.Notify() + หยุด ↓ Bridge.ExecuteCallback('CanEquipWings', source, 5) → เรียก callback ใน config.lua ของคุณ ↓ Callback คืนค่า false? → หยุด ↓ TriggerClientEvent(':spawn', source, 5) ↓ Client: สร้าง entity ของปีก ใช้สี 5 ↓ TriggerServerEvent(':sync:spawned', 5, 'idle') ↓ Server: บันทึกใน activeWings[source] Bridge.ExecuteCallback('OnWingsEquipped', source, 5) ↓ broadcastBucket(':sync:spawn', bucket, source, src, 5, 'idle') ↓ ผู้เล่นอื่นใน bucket เดียวกันเห็นปีกปรากฏ

Routing Buckets, StateBag และสถานะภายใน

ข้อมูลขั้นสูงเกี่ยวกับวิธีที่ resource จัดการการซิงโครไนซ์, ข้อขัดแย้งหลาย resource และสถานะภายใน

Routing Buckets

Resource รองรับ routing buckets ของ FiveM ผู้เล่นใน buckets ต่างกันจะไม่เห็นปีกของกันและกัน การซิงโครไนซ์ทั้งหมดถูกกรองตาม bucket

server (ภายใน)
-- ผู้เล่นแต่ละคนที่มีปีกจะมี bucket ของตนถูกติดตาม
activeWings[serverId] = {
    cor    = 5,
    state  = 'idle',
    bucket = 0,
}

-- Broadcasts ส่งเฉพาะผู้เล่นใน bucket เดียวกัน
broadcastBucket(eventName, bucket, excludeSource, ...)

-- เมื่อ bucket เปลี่ยน (ตรวจสอบทุก 2 วินาที):
-- ผู้เล่นเปลี่ยน bucket 0 → bucket 5
--   ผู้เล่นใน bucket 0: ได้รับ :sync:remove
--   ผู้เล่นใน bucket 5: ได้รับ :sync:spawn + สถานะ
ในทางปฏิบัติหากคุณใช้อพาร์ตเมนต์, instances หรือระบบใดก็ตามที่ใช้ routing buckets (เช่น: qb-apartments, esx_property) ปีกจะทำงานถูกต้องในแต่ละ instance โดยไม่รบกวนกัน

StateBag (หลาย Resource)

หากเซิร์ฟเวอร์มี resources ปีกหลายตัวจาก MathStore (เช่น: DragonWings + DemonWings) ระบบจะใช้ StateBag เพื่อหลีกเลี่ยงข้อขัดแย้ง มีเพียง resource เดียวที่สามารถมีปีกที่ใช้งานอยู่ต่อผู้เล่น

lua
-- คีย์ที่ใช้ใน state bag
WING_STATE_KEY = 'activeWingResource'

-- เมื่อสวมปีก server กำหนด:
Player(src).state:set(WING_STATE_KEY, RES_NAME, true)

-- หาก resource ปีกอื่นพยายามสวมผู้เล่นเดียวกัน:
-- ตรวจสอบว่ามี resource ที่ใช้งานอยู่หรือไม่
-- ถ้ามี สามารถลบตัวก่อนหน้าอัตโนมัติ

-- Client รับฟังการเปลี่ยนแปลงผ่าน:
AddStateBagChangeHandler(WING_STATE_KEY, ...)
